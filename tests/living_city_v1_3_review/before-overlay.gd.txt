func _label(parent: Node, value: String = "", font_size: int = 24) -> Label:
	var label := Label.new()
	label.text = value
	label.add_theme_font_size_override("font_size", font_size)
	if font_size >= 30: label.add_theme_font_override("font",CityStyle.Atlas.BOLD_FONT)
	label.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	label.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	parent.add_child(label)
	return label

func _button(parent: Node, value: String, action: Callable) -> Button:
	var button := Button.new()
	button.text = value
	CityStyle.button(button)
	parent.add_child(button)
	button.pressed.connect(action)
	return button

func _portrait(parent: Node, dimensions: Vector2) -> TextureRect:
	var image := TextureRect.new()
	image.custom_minimum_size = dimensions
	image.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	image.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	image.mouse_filter = Control.MOUSE_FILTER_IGNORE
	parent.add_child(image)
	return image

func _panel(parent: Node, dark: bool = false) -> PanelContainer:
	var panel := PanelContainer.new()
	panel.add_theme_stylebox_override("panel",CityStyle.panel(dark))
	parent.add_child(panel)
	return panel

func open(c: Node, city_id: String, task_kind: String) -> void:
	campaign = c; city = city_id; kind = task_kind
	personnel_mode = false
	choosing_officer = false
	preview_context = ""
	dismiss_confirmation.hide()
	selector.clear()
	refresh()
	show()
	move_to_front()
	choose_button.grab_focus()

func selected_id() -> String:
	return str(selector.get_item_metadata(selector.selected)) if selector.selected >= 0 else ""

func _texture(id: String) -> Texture2D:
	return campaign.map_area._get_portrait_texture(str(campaign.get_officer(id).get("name","")))

func _hero_texture(id: String, thumbnail: bool = false) -> Texture2D:
	# Independent face/shoulder framing; preserve crown and source aspect ratio.
	if campaign.get_officer(id).get("name","") != "선덕여왕": return _texture(id)
	var key := id+ (":thumb" if thumbnail else ":hero")
	if not hero_cache.has(key):
		var upper_body := AtlasTexture.new()
		upper_body.atlas = preload("res://assets/portraits/overlays/seondeok_queen_overlay.png")
		upper_body.region = Rect2(240,0,480,560) if thumbnail else Rect2(205,0,610,610)
		hero_cache[key] = upper_body
	return hero_cache[key]

func _role(id: String) -> String:
	if id.is_empty(): return "미배정"
	if campaign.officer_registry.posts.get("ruler:"+campaign.player_faction_id,"") == id:
		return "%s 군주" % campaign.player_faction
	if campaign.get_governor_id(city) == id: return "현재 도시 태수"
	return "현지 장수"

func _present_hero(id: String, caption: String) -> void:
	var person: Dictionary = campaign.get_officer(id)
	hero_caption.text = caption
	governor_label.text = str(person.get("name","공석" if not personnel_mode else "후보 없음"))
	governor_label.tooltip_text = governor_label.text
	governor_portrait.texture = _hero_texture(id)
	hero_ability.text = "정치 %d · 지력 %d" % [person.get("politics",0),person.get("intelligence",0)]
	hero_context.text = "%s\n개인 충성 %d" % [_role(id),person.get("loyalty",0)]
	var group: Dictionary = campaign.officer_registry.get("politics",{}).get("groups",{}).get(Politics.group(campaign.officer_registry,id),{})
	if not group.is_empty(): hero_context.text += "\n"+str(group.name)
	hero_context.tooltip_text = hero_context.text

func _update_detail_view(context: String) -> void:
	# Only a new city/mode/candidate/task resets reading position. Same-context
	# refreshes (including quote updates) preserve the player's scroll position.
	if preview_context != context:
		preview_context = context
		detail_scroll.scroll_vertical = 0
		_reset_detail_scroll.call_deferred(context)
	_update_more_label.call_deferred()

func _reset_detail_scroll(context: String) -> void:
	if preview_context == context: detail_scroll.scroll_vertical = 0

func _update_more_label() -> void:
	effect_heading.text = "임명 시 변화" if personnel_mode else "담당자 · 세부 정보"
	if details.size.y > detail_scroll.size.y: effect_heading.text += " · 상세 ↓"
	detail_scroll.tooltip_text = "긴 설명은 이 영역에서 스크롤할 수 있습니다."

func refresh() -> void:
	var previous := selected_id()
	var active: Dictionary = campaign.Domestic.active_job(campaign.strategy_state,city)
	if not personnel_mode and not active.is_empty(): previous = active.officer_id
	selector.clear()
	for id: String in campaign.get_city_officer_ids(city):
		var p: Dictionary = campaign.get_officer(id)
		selector.add_item("%s · 정치 %d / 지력 %d" % [p.name,p.politics,p.intelligence])
		selector.set_item_metadata(selector.item_count-1,id)
		if id == previous: selector.select(selector.item_count-1)
	_build_candidates()
	refresh_quote()

func _build_candidates() -> void:
	for child: Node in candidate_row.get_children():
		candidate_row.remove_child(child); child.queue_free()
	for n: int in range(selector.item_count):
		var id: String = str(selector.get_item_metadata(n))
		var p: Dictionary = campaign.get_officer(id)
		var card := VBoxContainer.new()
		card.custom_minimum_size.x = 185
		candidate_row.add_child(card)
		_portrait(card,Vector2(185,95)).texture = _hero_texture(id,true)
		var index := n
		var button := _button(card,p.name,func(): selector.select(index); refresh_quote())
		button.set_meta("officer_id",id)
		button.clip_text = true
		button.add_theme_font_size_override("font_size",20)
		button.tooltip_text = "%s · %s · 정치 %d / 지력 %d" % [p.name,_role(id),p.politics,p.intelligence]
	_build_officer_cards()

func _duties(person: Dictionary) -> String:
	var names: Array[String]=[]
	for duty: Dictionary in person.get("duties",[]):
		var task: String=str(duty.get("kind",""))
		names.append({"domestic":"내정 개발","production":"생산","diplomacy":"외교","army":"군사"}.get(task,task))
	return "현재 업무 · 없음" if names.is_empty() else "현재 업무 · "+", ".join(names)

func _build_officer_cards() -> void:
	for child: Node in picker_cards.get_children():
		picker_cards.remove_child(child); child.queue_free()
	if selector.item_count==0:
		_label(picker_cards,"이 도시에 배정할 후보가 없습니다.\n선택 취소로 돌아갈 수 있습니다.",24)
	for n: int in range(selector.item_count):
		var id: String=str(selector.get_item_metadata(n))
		var person: Dictionary=campaign.get_officer(id)
		var q: Dictionary=campaign.get_domestic_quote(city,kind,id)
		var selected: bool=id==selected_id()
		var card := _panel(picker_cards)
		card.set_meta("officer_id",id)
		card.add_theme_stylebox_override("panel",CityStyle.Atlas.panel(CityStyle.PAPER,CityStyle.GOLD if selected else CityStyle.LINE,3 if selected else 1,3))
		var row := HBoxContainer.new()
		row.add_theme_constant_override("separation",14)
		card.add_child(row)
		_portrait(row,Vector2(112,130)).texture=_hero_texture(id,true)
		var info := VBoxContainer.new()
		info.size_flags_horizontal=Control.SIZE_EXPAND_FILL
		row.add_child(info)
		var index := n
		var button := _button(info,("선택 중 · " if selected else "")+str(person.name),func(): _pick_officer(index))
		button.clip_text=true
		button.tooltip_text=str(person.name)+" · "+_role(id)
		button.set_meta("officer_id",id)
		CityStyle.button(button,selected)
		_label(info,"정치 %d · 지력 %d" % [person.politics,person.intelligence],23)
		var duty := _label(info,_duties(person),20)
		duty.max_lines_visible=2; duty.tooltip_text=duty.text
		var reason := _label(info,"배정 가능" if q.ok else "배정 불가 · "+str(q.reason),20)
		reason.add_theme_color_override("font_color",CityStyle.INK if q.ok else CityStyle.RED)
		card.set_meta("reason",reason.text)

func _open_officers() -> void:
	if personnel_mode: return
	choosing_officer=true
	_build_officer_cards()
	picker_scroll.scroll_vertical=0
	refresh_quote()
	cancel_button.grab_focus()

func _pick_officer(index: int) -> void:
	selector.select(index)
	choosing_officer=false
	refresh_quote()
	choose_button.grab_focus()

func back() -> void:
	if choosing_officer:
		choosing_officer=false
		refresh_quote()
		choose_button.grab_focus()
	else: hide()

func refresh_quote() -> void:
	var province: Dictionary = campaign.provinces[city]
	var g: String = campaign.get_governor_id(city)
	var p: Dictionary = campaign.get_officer(g)
	header.text = "%s · %s  |  %d년 %d월     국고 %d  |  도시 군량 %d" % [campaign.player_faction,province.name,campaign.year,campaign.month,campaign.gold,province.food_stock]
	city_title.text = province.name+" 내정"
	city_art.visible = city == "geumseong"
	art_note.text = "사람이 모여, 나라가 살아 숨 쉬는 서라벌" if city == "geumseong" else "도시 전경 준비 중 · 실제 도시 현황"
	_present_hero(g,"현재 태수")
	var operation: Dictionary = campaign.city_operation_quote(city)
	hero_context.text += "\n운영 효과 +%.1f%%" % ((float(operation.governor.multiplier)-1.0)*100.0)
	current_summary.text = "현재 태수 · %s\n현재 월 세입 %d · 운영 +%.1f%%" % [p.get("name","공석"),operation.tax,(float(operation.governor.multiplier)-1.0)*100.0]
	current_summary.tooltip_text = campaign.get_city_operation_text(city)
	current_summary.get_parent().visible = personnel_mode or choosing_officer
	city_stats.text = "민간 인구 %d  ·  치안 %d\n농업 %d  ·  상업 %d  ·  월 세입 %d" % [province.population,province.public_order,province.agriculture,province.commerce,campaign.city_operation_quote(city).tax]
	var lines: Array[String] = []
	for job: Dictionary in campaign.strategy_state.get("domestic",{}).get("jobs",{}).values():
		if job.city_id != city: continue
		var job_name: String = {"agriculture":"농업 개발","commerce":"상업 개발"}.get(job.kind,job.get("name",job.kind))
		var job_status: String = {"pending":"진행 중","paused":"중지","completed":"완료","cancelled":"취소","interrupted":"중단"}.get(job.status,job.status)
		var reason: String = str(job.get("reason",""))
		if reason.is_empty() and job.has("due_month"):
			var due: int = int(job.due_month)
			reason = "완료 예정 %d년 %d월" % [int((due-1)/12.0),(due-1)%12+1]
		elif reason.is_empty() and job.has("progress"):
			reason = "진행 %d / %d" % [job.progress,job.get("required",0)]
		lines.append("%s · %s · %s\n%s" % [job_name,campaign.get_officer(job.officer_id).get("name",job.officer_id),job_status,reason])
	jobs_label.text = "진행 중인 개발 업무가 없습니다." if lines.is_empty() else "\n".join(lines)
	candidate_scroll.visible = personnel_mode
	tasks_row.visible = not personnel_mode and not choosing_officer
	governor_portrait.get_parent().visible = not choosing_officer
	selector.hide()
	choose_button.visible = not personnel_mode and not choosing_officer
	officer_row.visible = false
	quote_summary.visible = not personnel_mode and not choosing_officer
	picker_scroll.visible = choosing_officer
	detail_scroll.visible = not choosing_officer
	effect_heading.visible = not choosing_officer
	execute_button.visible = not choosing_officer
	var selected_person: Dictionary = campaign.get_officer(selected_id())
	officer_portrait.texture = _hero_texture(selected_id(),true)
	officer_summary.text = "담당자  %s\n위치 %s · %s" % [selected_person.get("name","후보 없음"),campaign.provinces.get(selected_person.get("location",""),{}).get("name","미배치"),_role(selected_id())]
	for duty: Dictionary in selected_person.get("duties",[]): officer_summary.text += " · "+str(duty.get("kind",""))
	governor_button.text = "내정으로" if personnel_mode else "태수 변경"
	for key: String in navigation: CityStyle.button(navigation[key],key == ("personnel" if personnel_mode else "domestic"))
	for key: String in task_buttons: CityStyle.button(task_buttons[key],not personnel_mode and key==kind)
	for card: Node in candidate_row.get_children():
		var button: Button = card.get_child(1)
		var selected: bool = button.get_meta("officer_id") == selected_id()
		CityStyle.button(button,selected)
		button.text = ("선택 · " if selected else "")+str(campaign.get_officer(button.get_meta("officer_id")).name)
	dismiss_button.visible = personnel_mode
	dismiss_button.text = "%s 해임" % p.get("name","태수")
	dismiss_button.tooltip_text = "현재 태수 %s 해임 · 확인 후 기존 인계 규칙 적용" % p.get("name","공석")
	dismiss_button.disabled = g.is_empty()
	cancel_button.text = "선택 취소" if personnel_mode else "업무 취소"
	if choosing_officer:
		title.text="담당자 선택"
		status.text="카드 선택 후 개발 지시를 눌러야 접수됩니다."
		cancel_button.text="선택 취소"
		cancel_button.disabled=false
		return
	if personnel_mode:
		effect_heading.text = "임명 시 변화"
		_personnel_quote()
		_update_detail_view(city+":personnel:"+selected_id())
		return
	var q: Dictionary = campaign.get_domestic_quote(city,kind,selected_id())
	var active: Dictionary = campaign.Domestic.active_job(campaign.strategy_state,city)
	title.text = "농업 개발" if kind=="agriculture" else "상업 개발"
	effect_heading.text = "개발 견적 · 선택 담당자 기준"
	var due: int = int(active.get("due_month",q.due_month))
	selector.disabled = not active.is_empty()
	choose_button.text="담당자 · %s  ›" % selected_person.get("name","후보 없음")
	choose_button.clip_text=true
	choose_button.tooltip_text=choose_button.text
	choose_button.disabled=not active.is_empty()
	quote_values[0].text="금 %d" % q.cost
	quote_values[1].text="%d. %d월" % [int((due-1)/12.0),(due-1)%12+1]
	quote_values[1].tooltip_text="%d년 %d월 완료 예정 · 월 진행 %d회" % [int((due-1)/12.0),(due-1)%12+1,campaign.Domestic.MONTHS]
	quote_values[2].text=("+%d" % active.get("planned_gain",q.gain)) if q.ok or not active.is_empty() or q.has("efficiency") else "—"
	details.text = "%s\n정치 %d · 지력 %d\n기간 · 월 진행 %d회 / 성과 상한 %d\n%s\n\n현재 태수의 운영 효과 (개발 성과와 별도)\n%s" % [officer_summary.text,selected_person.get("politics",0),selected_person.get("intelligence",0),campaign.Domestic.MONTHS,campaign.Domestic.MAX_DEVELOPMENT,_duties(selected_person),campaign.get_city_operation_text(city)]
	if not lines.is_empty(): details.text += "\n\n"+lines.back()
	status.text = "담당자 배정 가능" if q.ok else str(q.reason)
	execute_button.text = "개발 지시 · 금 %d" % q.cost
	execute_button.disabled = not q.ok or busy
	cancel_button.disabled = active.is_empty()
	status.tooltip_text = status.text
	_update_detail_view(city+":"+kind+":"+selected_id())

func _personnel_quote() -> void:
	title.text = "태수 임명"
	cancel_button.disabled = false
	execute_button.text = "임명 확정"
	var id := selected_id()
	var q: Dictionary = Personnel.quote(campaign,campaign.player_faction_id,"governor",city,id)
	execute_button.disabled = not q.ok or busy
	status.text = "후보가 없습니다. 선택을 취소할 수 있습니다." if id.is_empty() else ("확정 전에는 직책·국고가 변경되지 않습니다." if q.ok else str(q.reason))
	status.tooltip_text = status.text
	_present_hero(id,"선택 후보")
	var person: Dictionary = campaign.get_officer(id)
	var old_id: String = campaign.get_governor_id(city)
	var old_name: String = campaign.get_officer(old_id).get("name","공석")
	details.text = "현재 %s → 후보 %s" % [old_name,person.get("name","없음")]
	if not q.ok:
		details.text += "\n"+str(q.reason)
		return
	var transfer: Dictionary = campaign.Power.quote(campaign,{"kind":"governor","target":city,"officer_id":id,"faction_id":campaign.player_faction_id})
	if not transfer.ok:
		execute_button.disabled = true
		status.text = transfer.reason
		status.tooltip_text = status.text
		details.text += "\n"+str(transfer.reason)
		return
	var difference: int = int(q.ability)-int(q.old_ability)
	details.text += "\n태수 정치 %d → %d (차이 %s%d)" % [q.old_ability,q.ability,"+" if difference>=0 else "",difference]
	if transfer.get("required",false):
		status.text = "권력 인계 협의가 필요합니다."
		execute_button.text = "인계 조건 확인"
		details.text += "\n"+campaign.Power.describe(campaign,transfer)
	else:
		details.text += "\n현지 임명 · 기존 개발 업무 유지"
		for reaction: Dictionary in q.reactions:
			var loyalty_delta: int = int(reaction.loyalty_after)-int(reaction.loyalty_before)
			var cooperation_delta: int = int(reaction.cooperation_after)-int(reaction.cooperation_before)
			details.text += "\n%s 개인 충성 %d → %d (%s%d)\n집단 협력 %d → %d (%s%d)" % [campaign.get_officer(reaction.officer_id).name,reaction.loyalty_before,reaction.loyalty_after,"+" if loyalty_delta>=0 else "",loyalty_delta,reaction.cooperation_before,reaction.cooperation_after,"+" if cooperation_delta>=0 else "",cooperation_delta]
		if q.reactions.is_empty(): details.text += "\n공통 견적의 개인 충성·집단 협력 변화 없음"
		for gid: String in q.before.groups:
			var group: Dictionary = campaign.officer_registry.politics.groups[gid]
			var before: float = q.before.groups[gid].influence
			var after: float = q.after.groups[gid].influence
			details.text += "\n%s 집단 영향력 %.1f → %.1f (%s%.1f)" % [group.name,before,after,"+" if after>=before else "",after-before]
	if _role(id).ends_with("군주"):
		details.text += "\n군주 후보: 기존 공통 임명 검증에서 태수 겸임을 허용합니다. 군주 직책은 유지됩니다."
	details.text += "\n세입·수확 예상 증감은 제공되지 않습니다.\n임명 후 실제 운영 수치를 갱신합니다."
	for duty: Dictionary in person.get("duties",[]): details.text += "\n후보의 현재 업무: "+str(duty.get("kind",""))


func _execute() -> void:
	if busy or choosing_officer or execute_button.disabled or dismiss_confirmation.visible: return
	busy = true
	var result: Dictionary
	if personnel_mode: result = Personnel.appoint(campaign,campaign.player_faction_id,"governor",city,selected_id())
	else: result = campaign.start_domestic(city,kind,selected_id())
	busy = false
	campaign.update_top_bar()
	refresh()
	status.text = str(result.get("reason","")) if not result.ok else ("태수 임명 완료" if personnel_mode else "개발 접수 완료")

func _cancel() -> void:
	if choosing_officer: back(); return
	if personnel_mode: _set_mode(false); return
	var job: Dictionary = campaign.Domestic.active_job(campaign.strategy_state,city)
	if not job.is_empty(): campaign.cancel_domestic(job.id)
	refresh()

func _dismiss() -> void:
	dismiss_target = campaign.get_governor_id(city)
	if dismiss_target.is_empty(): return
	dismiss_confirmation.dialog_text = "현재 태수 %s을(를) 해임하시겠습니까?\n확정 시 기존 권력 인계 규칙을 적용합니다." % campaign.get_officer(dismiss_target).name
	dismiss_confirmation.popup_centered(Vector2i(650,190))

func _confirm_dismiss() -> void:
	if dismiss_target.is_empty() or campaign.get_governor_id(city) != dismiss_target:
		refresh()
		status.text = "현재 태수가 바뀌었습니다. 다시 확인하세요."
		return
	var success: bool = campaign.dismiss_governor(city)
	dismiss_target = ""
	refresh()
	status.text = "태수 해임 완료" if success else "해임 제한 · 권력 인계 현황을 확인하세요."

func _set_mode(value: bool) -> void:
	choosing_officer=false
	personnel_mode = value
	refresh()

func _task(value: String) -> void:
	if value == "industry": _navigate("industry"); return
	kind = value; personnel_mode = false; choosing_officer=false; refresh()

func _navigate(target: String) -> void:
	match target:
		"domestic": _set_mode(false)
		"personnel": _set_mode(true)
		"industry": hide(); campaign._on_city_card_production_requested(city)
		"politics": hide(); campaign.open_politics()
		"diplomacy": hide(); campaign._open_diplomacy()
		"map": hide()

func _next_month() -> void:
	# The existing turn handler retains event, ending and transition locks.
	if busy: return
	hide()
	campaign._on_end_turn_button_pressed()
